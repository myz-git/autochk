select sql_id,round(sum(SHARABLE_MEM/1024/1024),3) SHARED_MB
from V$SQLAREA  group by sql_id
having sum(SHARABLE_MEM/1024/1024) <0 or sum(SHARABLE_MEM/1024/1024) >300
order by 2 ;
